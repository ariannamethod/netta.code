size = 7
canvas = [['.' for column in range(size)] for row in range(size)]
pending = [(size // 2, size // 2)]
while pending:
    row, column = pending.pop()
    if abs(row - size // 2) + abs(column - size // 2) <= size // 2 and canvas[row][column] == '.':
        canvas[row][column] = '#'
        for next_row, next_column in [(row - 1, column), (row + 1, column), (row, column - 1), (row, column + 1)]:
            if 0 <= next_row < size and 0 <= next_column < size:
                pending.append((next_row, next_column))
for row in canvas:
    print(''.join(row))
