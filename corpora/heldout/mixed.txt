# === PROGRAM ===
limit = 7 * 3
prime = [True] * limit
prime[0] = False
prime[1] = False
for number in range(2, limit):
    if prime[number]:
        for multiple in range(number * number, limit, number):
            prime[multiple] = False
print([number for number in range(limit) if prime[number]])

values = sorted([4, 2, 9, 3, 2, 7, 11, 5, 6, 8])
left = []
right = []
for value in reversed(values):
    if sum(left) <= sum(right):
        left.append(value)
    else:
        right.append(value)
print(left)
print(right)
