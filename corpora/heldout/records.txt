rows = [['river', 3], ['stone', 9], ['river', 5], ['leaf', 12]]
result = []
for name, value in rows:
    dominated = False
    for other, score in rows:
        if len(other) <= len(name) and score >= value and (len(other) < len(name) or score > value):
            dominated = True
    if not dominated:
        result.append([name, value])
print(result)

rows = [['river', 3], ['stone', 9], ['river', 5], ['leaf', 12]]
total = sum(value for name, value in rows)
result = []
for target in range(0, total, 2):
    running = 0
    for name, value in rows:
        running += value
        if target < running:
            result.append(name)
            break
print(result)
