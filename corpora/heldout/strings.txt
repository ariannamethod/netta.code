words = ['river', 'stone', 'river', 'leaf', 'moon']
left = words[0]
right = words[1]
previous = list(range(len(right) + 1))
for row, first in enumerate(left, 1):
    current = [row]
    for column, second in enumerate(right, 1):
        current.append(min(current[-1] + 1, previous[column] + 1, previous[column - 1] + (first != second)))
    previous = current
print(previous[-1])

text = '(' + 'still waters turn slowly' + '(' + 'still waters turn slowly' + '))'
depth = 0
maximum = 0
for letter in text:
    if letter == '(':
        depth += 1
        maximum = max(maximum, depth)
    elif letter == ')':
        depth -= 1
print(maximum)

words = ['river', 'stone', 'river', 'leaf', 'moon']
root = dict()
for word in words:
    node = root
    for letter in word:
        if letter not in node:
            node[letter] = dict()
        node = node[letter]
    node['end'] = True
print(root)

words = sorted(set(['river', 'stone', 'river', 'leaf', 'moon']))
result = []
for word in words:
    for size in range(1, len(word) + 1):
        prefix = word[:size]
        if sum(other.startswith(prefix) for other in words) == 1:
            result.append(prefix)
            break
print(result)
